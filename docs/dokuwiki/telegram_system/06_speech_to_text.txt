===== 📋 Visão Geral =====

Sistema de conversão de voz para texto que processa mensagens de voz do Telegram e as converte em texto para processamento posterior. Integra com APIs de speech-to-text e gerencia cache de arquivos de áudio.

===== 🚀 Comando de Implementação =====

<code bash>
implementar sistema speech to text telegram
</code>

===== ⚙️ Pré-requisitos =====

  - **Sistema de Logging** implementado
  - **Sistema de Rastreamento de Fluxo** implementado
  - **Sistema de Canais de Notificação** implementado
  - **Sistema de Validação e Middleware** implementado
  - **Services Core do Telegram** implementados
  - **Laravel 12** instalado
  - **PHP 8.2+** configurado
  - **FFmpeg** para processamento de áudio
  - **API de Speech-to-Text** (OpenAI Whisper, Google Speech, etc.)

<code php>
<?php

namespace App\Services;

use App\Contracts\LoggingServiceInterface;
use App\Services\Telegram\TelegramFileService;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Facades\Http;

class SpeechToTextService
{
    private string $storagePath;
    private int $maxFileSize;
    private int $maxDuration;

    public function __construct(
        private LoggingServiceInterface $loggingService,
        private TelegramFileService $telegramFileService
    ) {
        $this->storagePath = config('speech.storage_path', 'speech-temp');
        $this->maxFileSize = config('speech.max_file_size', 20 * 1024 * 1024); // 20MB
        $this->maxDuration = config('speech.max_duration', 300); // 5 minutes
    }

    /**
     * Convert voice file from Telegram to text
     */
    public function convertVoiceToText(string $fileId): ?string
    {
        try {
            $this->loggingService->logTelegramEvent('speech_conversion_started', [
                'file_id' => $fileId
            ], 'info');

            // Download file from Telegram
            $filePath = $this->telegramFileService->downloadFile($fileId);
            
            if (!$filePath) {
                $this->loggingService->logTelegramEvent('speech_file_download_failed', [
                    'file_id' => $fileId
                ], 'error');
                return null;
            }

            // Validate file
            if (!$this->validateAudioFile($filePath)) {
                $this->loggingService->logTelegramEvent('speech_file_validation_failed', [
                    'file_id' => $fileId,
                    'file_path' => $filePath
                ], 'error');
                return null;
            }

            // Convert to text using configured provider
            $text = $this->performSpeechToTextConversion($filePath);

            // Clean up temporary file
            $this->cleanupTempFile($filePath);

            if ($text) {
                $this->loggingService->logTelegramEvent('speech_conversion_successful', [
                    'file_id' => $fileId,
                    'text_length' => strlen($text)
                ], 'info');
            } else {
                $this->loggingService->logTelegramEvent('speech_conversion_failed', [
                    'file_id' => $fileId
                ], 'error');
            }

            return $text;

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'convert_voice_to_text',
                'file_id' => $fileId
            ]);

            return null;
        }
    }

    /**
     * Perform speech-to-text conversion
     */
    private function performSpeechToTextConversion(string $filePath): ?string
    {
        $provider = config('speech.provider', 'openai');

        return match ($provider) {
            'openai' => $this->convertWithOpenAI($filePath),
            'google' => $this->convertWithGoogle($filePath),
            'azure' => $this->convertWithAzure($filePath),
            default => $this->convertWithOpenAI($filePath)
        };
    }

    /**
     * Convert using OpenAI Whisper API
     */
    private function convertWithOpenAI(string $filePath): ?string
    {
        try {
            $apiKey = config('speech.openai.api_key');
            
            if (!$apiKey) {
                $this->loggingService->logTelegramEvent('openai_api_key_missing', [], 'error');
                return null;
            }

            $response = Http::withHeaders([
                'Authorization' => 'Bearer ' . $apiKey,
            ])->attach('file', file_get_contents($filePath), basename($filePath))
            ->post('https://api.openai.com/v1/audio/transcriptions', [
                'model' => config('speech.openai.model', 'whisper-1'),
                'language' => config('speech.openai.language', 'pt'),
                'response_format' => 'text'
            ]);

            if ($response->successful()) {
                return trim($response->body());
            }

            $this->loggingService->logTelegramEvent('openai_api_error', [
                'status' => $response->status(),
                'error' => $response->body()
            ], 'error');

            return null;

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'openai_speech_conversion'
            ]);
            return null;
        }
    }

    /**
     * Convert using Google Speech API
     */
    private function convertWithGoogle(string $filePath): ?string
    {
        try {
            $apiKey = config('speech.google.api_key');
            
            if (!$apiKey) {
                $this->loggingService->logTelegramEvent('google_api_key_missing', [], 'error');
                return null;
            }

            // Convert to base64
            $audioContent = base64_encode(file_get_contents($filePath));

            $response = Http::post("https://speech.googleapis.com/v1/speech:recognize?key={$apiKey}", [
                'config' => [
                    'encoding' => 'OGG_OPUS',
                    'sampleRateHertz' => 48000,
                    'languageCode' => config('speech.google.language', 'pt-BR')
                ],
                'audio' => [
                    'content' => $audioContent
                ]
            ]);

            if ($response->successful()) {
                $data = $response->json();
                
                if (isset($data['results'][0]['alternatives'][0]['transcript'])) {
                    return $data['results'][0]['alternatives'][0]['transcript'];
                }
            }

            $this->loggingService->logTelegramEvent('google_api_error', [
                'status' => $response->status(),
                'error' => $response->body()
            ], 'error');

            return null;

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'google_speech_conversion'
            ]);
            return null;
        }
    }

    /**
     * Convert using Azure Speech Services
     */
    private function convertWithAzure(string $filePath): ?string
    {
        try {
            $apiKey = config('speech.azure.api_key');
            $region = config('speech.azure.region');
            
            if (!$apiKey || !$region) {
                $this->loggingService->logTelegramEvent('azure_api_config_missing', [], 'error');
                return null;
            }

            $response = Http::withHeaders([
                'Ocp-Apim-Subscription-Key' => $apiKey,
                'Content-Type' => 'audio/ogg; codecs=opus'
            ])->post("https://{$region}.stt.speech.microsoft.com/speech/recognition/conversation/cognitiveservices/v1", [
                'language' => config('speech.azure.language', 'pt-BR'),
                'format' => 'detailed'
            ]);

            if ($response->successful()) {
                $data = $response->json();
                
                if (isset($data['DisplayText'])) {
                    return $data['DisplayText'];
                }
            }

            $this->loggingService->logTelegramEvent('azure_api_error', [
                'status' => $response->status(),
                'error' => $response->body()
            ], 'error');

            return null;

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'azure_speech_conversion'
            ]);
            return null;
        }
    }

    /**
     * Validate audio file
     */
    private function validateAudioFile(string $filePath): bool
    {
        try {
            // Check if file exists
            if (!file_exists($filePath)) {
                return false;
            }

            // Check file size
            $fileSize = filesize($filePath);
            if ($fileSize > $this->maxFileSize) {
                $this->loggingService->logTelegramEvent('audio_file_too_large', [
                    'file_size' => $fileSize,
                    'max_size' => $this->maxFileSize
                ], 'warning');
                return false;
            }

            // Check file duration using FFmpeg
            $duration = $this->getAudioDuration($filePath);
            if ($duration > $this->maxDuration) {
                $this->loggingService->logTelegramEvent('audio_file_too_long', [
                    'duration' => $duration,
                    'max_duration' => $this->maxDuration
                ], 'warning');
                return false;
            }

            return true;

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'validate_audio_file',
                'file_path' => $filePath
            ]);
            return false;
        }
    }

    /**
     * Get audio file duration using FFmpeg
     */
    private function getAudioDuration(string $filePath): int
    {
        try {
            $command = sprintf(
                'ffprobe -v quiet -show_entries format=duration -of csv="p=0" %s',
                escapeshellarg($filePath)
            );

            $output = shell_exec($command);
            
            return (int) round((float) $output);

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'get_audio_duration',
                'file_path' => $filePath
            ]);
            return 0;
        }
    }

    /**
     * Clean up temporary file
     */
    private function cleanupTempFile(string $filePath): void
    {
        try {
            if (file_exists($filePath)) {
                unlink($filePath);
            }
        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'cleanup_temp_file',
                'file_path' => $filePath
            ]);
        }
    }

    /**
     * Test speech-to-text service
     */
    public function testService(): array
    {
        try {
            $provider = config('speech.provider', 'openai');
            $apiKey = config("speech.{$provider}.api_key");

            return [
                'success' => true,
                'provider' => $provider,
                'api_key_configured' => !empty($apiKey),
                'storage_path' => $this->storagePath,
                'max_file_size' => $this->maxFileSize,
                'max_duration' => $this->maxDuration
            ];

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'test_speech_service'
            ]);

            return [
                'success' => false,
                'error' => $e->getMessage()
            ];
        }
    }
}

=== Passo 2: Criar TelegramFileService ===

<code php>
<?php

namespace App\Services\Telegram;

use App\Contracts\LoggingServiceInterface;
use App\Services\Channels\TelegramChannel;
use Illuminate\Support\Facades\Storage;

class TelegramFileService
{
    private string $storagePath;

    public function __construct(
        private LoggingServiceInterface $loggingService,
        private TelegramChannel $telegramChannel
    ) {
        $this->storagePath = config('speech.storage_path', 'speech-temp');
    }

    /**
     * Download file from Telegram
     */
    public function downloadFile(string $fileId): ?string
    {
        try {
            $this->loggingService->logTelegramEvent('telegram_file_download_started', [
                'file_id' => $fileId
            ], 'info');

            // Get file info from Telegram
            $fileInfo = $this->telegramChannel->getFile($fileId);
            
            if (!$fileInfo['success']) {
                $this->loggingService->logTelegramEvent('telegram_file_info_failed', [
                    'file_id' => $fileId,
                    'error' => $fileInfo['error']
                ], 'error');
                return null;
            }

            $filePath = $fileInfo['file_path'];
            $botToken = config('services.telegram.bot_token');
            
            // Download file from Telegram servers
            $fileUrl = "https://api.telegram.org/file/bot{$botToken}/{$filePath}";
            
            $response = \Illuminate\Support\Facades\Http::get($fileUrl);
            
            if (!$response->successful()) {
                $this->loggingService->logTelegramEvent('telegram_file_download_failed', [
                    'file_id' => $fileId,
                    'file_url' => $fileUrl,
                    'status' => $response->status()
                ], 'error');
                return null;
            }

            // Save to temporary file
            $tempFileName = "telegram_{$fileId}_" . time() . ".ogg";
            $tempFilePath = storage_path("app/{$this->storagePath}/{$tempFileName}");
            
            // Ensure directory exists
            $directory = dirname($tempFilePath);
            if (!is_dir($directory)) {
                mkdir($directory, 0755, true);
            }

            file_put_contents($tempFilePath, $response->body());

            $this->loggingService->logTelegramEvent('telegram_file_download_successful', [
                'file_id' => $fileId,
                'temp_file' => $tempFilePath,
                'file_size' => filesize($tempFilePath)
            ], 'info');

            return $tempFilePath;

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'download_telegram_file',
                'file_id' => $fileId
            ]);
            return null;
        }
    }

    /**
     * Get file information from Telegram
     */
    public function getFileInfo(string $fileId): array
    {
        try {
            return $this->telegramChannel->getFile($fileId);
        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'get_telegram_file_info',
                'file_id' => $fileId
            ]);
            
            return [
                'success' => false,
                'error' => $e->getMessage()
            ];
        }
    }

    /**
     * Clean up old temporary files
     */
    public function cleanupOldFiles(int $maxAge = 3600): int
    {
        try {
            $tempDir = storage_path("app/{$this->storagePath}");
            
            if (!is_dir($tempDir)) {
                return 0;
            }

            $files = glob($tempDir . '/telegram_*');
            $cleanedCount = 0;
            $cutoffTime = time() - $maxAge;

            foreach ($files as $file) {
                if (filemtime($file) < $cutoffTime) {
                    unlink($file);
                    $cleanedCount++;
                }
            }

            if ($cleanedCount > 0) {
                $this->loggingService->logTelegramEvent('telegram_temp_files_cleaned', [
                    'cleaned_count' => $cleanedCount,
                    'max_age_seconds' => $maxAge
                ], 'info');
            }

            return $cleanedCount;

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'cleanup_telegram_temp_files'
            ]);
            return 0;
        }
    }
}
