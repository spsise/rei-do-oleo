=== Passo 2: Criar TelegramConfigController ===

<code php>
<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Services\Telegram\TelegramWebhookService;
use App\Contracts\LoggingServiceInterface;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class TelegramConfigController extends Controller
{
    public function __construct(
        private LoggingServiceInterface $loggingService,
        private TelegramWebhookService $webhookService
    ) {}

    /**
     * Setup webhook
     */
    public function setupWebhook(Request $request): JsonResponse
    {
        try {
            $request->validate([
                'webhook_url' => 'required|url|max:255',
                'secret_token' => 'nullable|string|max:255'
            ]);

            $webhookUrl = $request->input('webhook_url');
            $secretToken = $request->input('secret_token', config('services.telegram.webhook_secret'));

            $result = $this->webhookService->setWebhook($webhookUrl, $secretToken);

            if ($result['success']) {
                $this->loggingService->logBusinessOperation('webhook_configured_successfully', [
                    'webhook_url' => $webhookUrl,
                    'has_secret' => !empty($secretToken)
                ], 'success');
            }

            return response()->json($result);

        } catch (\Illuminate\Validation\ValidationException $e) {
            $this->loggingService->logTelegramEvent('webhook_config_validation_failed', [
                'errors' => $e->errors()
            ], 'error');

            return response()->json([
                'success' => false,
                'error' => 'Validation failed',
                'errors' => $e->errors()
            ], 422);

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'setup_webhook'
            ]);

            return response()->json([
                'success' => false,
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Get webhook info
     */
    public function getWebhookInfo(): JsonResponse
    {
        try {
            $result = $this->webhookService->getWebhookInfo();

            $this->loggingService->logBusinessOperation('webhook_info_retrieved', [
                'success' => $result['success']
            ], 'info');

            return response()->json($result);

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'get_webhook_info'
            ]);

            return response()->json([
                'success' => false,
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Delete webhook
     */
    public function deleteWebhook(): JsonResponse
    {
        try {
            $result = $this->webhookService->deleteWebhook();

            if ($result['success']) {
                $this->loggingService->logBusinessOperation('webhook_deleted_successfully', [], 'success');
            }

            return response()->json($result);

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'delete_webhook'
            ]);

            return response()->json([
                'success' => false,
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Get bot info
     */
    public function getBotInfo(): JsonResponse
    {
        try {
            $result = $this->webhookService->getBotInfo();

            $this->loggingService->logBusinessOperation('bot_info_retrieved', [
                'success' => $result['success']
            ], 'info');

            return response()->json($result);

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'get_bot_info'
            ]);

            return response()->json([
                'success' => false,
                'error' => $e->getMessage()
            ], 500);
        }
    }
}
</code>

=== Passo 6: Criar Comando Artisan para Setup ===

<code php>
<?php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use App\Services\Telegram\TelegramWebhookService;
use App\Contracts\LoggingServiceInterface;

class TelegramSetupCommand extends Command
{
    protected $signature = 'telegram:setup {--webhook-url=} {--force}';
    protected $description = 'Setup Telegram bot webhook and configuration';

    public function __construct(
        private LoggingServiceInterface $loggingService,
        private TelegramWebhookService $webhookService
    ) {
        parent::__construct();
    }

    public function handle(): int
    {
        try {
            $this->info('🤖 Configurando Telegram Bot...');

            // Check bot connection
            $botInfo = $this->webhookService->getBotInfo();
            
            if (!$botInfo['success']) {
                $this->error('❌ Falha ao conectar com o bot: ' . $botInfo['error']);
                return 1;
            }

            $this->info('✅ Bot conectado: ' . $botInfo['bot_info']['first_name']);

            // Setup webhook
            $webhookUrl = $this->option('webhook-url') ?: $this->ask('URL do Webhook');
            
            if (!$webhookUrl) {
                $this->error('❌ URL do webhook é obrigatória');
                return 1;
            }

            $result = $this->webhookService->setWebhook($webhookUrl);

            if ($result['success']) {
                $this->info('✅ Webhook configurado com sucesso!');
                $this->info('🔗 URL: ' . $webhookUrl);
            } else {
                $this->error('❌ Falha ao configurar webhook: ' . $result['error']);
                return 1;
            }

            // Health check
            $health = $this->webhookService->healthCheck();
            
            if ($health['success']) {
                $this->info('✅ Sistema funcionando corretamente!');
                $this->info('🤖 Bot: ' . $health['bot_info']['first_name']);
                $this->info('🔗 Webhook: ' . ($health['webhook_configured'] ? 'Configurado' : 'Não configurado'));
            }

            $this->loggingService->logBusinessOperation('telegram_setup_completed', [
                'webhook_url' => $webhookUrl,
                'bot_name' => $botInfo['bot_info']['first_name']
            ], 'success');

            return 0;

        } catch (\Exception $e) {
            $this->error('❌ Erro durante configuração: ' . $e->getMessage());
            
            $this->loggingService->logException($e, [
                'operation' => 'telegram_setup_command'
            ]);

            return 1;
        }
    }
}
</code>
