<?php

namespace App\Services\Telegram\Commands\Advanced;

use App\Contracts\LoggingServiceInterface;
use Illuminate\Support\Facades\Cache;

class UserStateManager
{
    private string $cachePrefix = 'telegram_user_state_';
    private int $defaultTtl = 3600; // 1 hour

    public function __construct(
        private LoggingServiceInterface $loggingService
    ) {}

    /**
     * Set user state
     */
    public function setUserState(int $userId, array $state): void
    {
        try {
            $cacheKey = $this->getCacheKey($userId);
            
            Cache::put($cacheKey, $state, $this->defaultTtl);
            
            $this->loggingService->logTelegramEvent('user_state_updated', [
                'user_id' => $userId,
                'state_keys' => array_keys($state)
            ], 'info');

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'set_user_state',
                'user_id' => $userId
            ]);
        }
    }

    /**
     * Get user state
     */
    public function getUserState(int $userId): array
    {
        try {
            $cacheKey = $this->getCacheKey($userId);
            return Cache::get($cacheKey, []);
        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'get_user_state',
                'user_id' => $userId
            ]);
            return [];
        }
    }

    /**
     * Update specific state key
     */
    public function updateUserState(int $userId, string $key, mixed $value): void
    {
        try {
            $state = $this->getUserState($userId);
            $state[$key] = $value;
            $this->setUserState($userId, $state);
        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'update_user_state',
                'user_id' => $userId,
                'key' => $key
            ]);
        }
    }

    /**
     * Get specific state value
     */
    public function getUserStateValue(int $userId, string $key, mixed $default = null): mixed
    {
        $state = $this->getUserState($userId);
        return $state[$key] ?? $default;
    }

    /**
     * Clear user state
     */
    public function clearUserState(int $userId): void
    {
        try {
            $cacheKey = $this->getCacheKey($userId);
            Cache::forget($cacheKey);
            
            $this->loggingService->logTelegramEvent('user_state_cleared', [
                'user_id' => $userId
            ], 'info');
        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'clear_user_state',
                'user_id' => $userId
            ]);
        }
    }

    /**
     * Set user in menu
     */
    public function setUserInMenu(int $userId, string $menuName, array $menuData = []): void
    {
        $this->updateUserState($userId, 'current_menu', $menuName);
        $this->updateUserState($userId, 'menu_data', $menuData);
        $this->updateUserState($userId, 'menu_timestamp', now()->timestamp);
    }

    /**
     * Get user current menu
     */
    public function getUserCurrentMenu(int $userId): ?string
    {
        return $this->getUserStateValue($userId, 'current_menu');
    }

    /**
     * Clear user menu
     */
    public function clearUserMenu(int $userId): void
    {
        $this->updateUserState($userId, 'current_menu', null);
        $this->updateUserState($userId, 'menu_data', []);
    }

    /**
     * Set user context
     */
    public function setUserContext(int $userId, string $context, array $contextData = []): void
    {
        $this->updateUserState($userId, 'context', $context);
        $this->updateUserState($userId, 'context_data', $contextData);
    }

    /**
     * Get user context
     */
    public function getUserContext(int $userId): ?string
    {
        return $this->getUserStateValue($userId, 'context');
    }

    /**
     * Clear user context
     */
    public function clearUserContext(int $userId): void
    {
        $this->updateUserState($userId, 'context', null);
        $this->updateUserState($userId, 'context_data', []);
    }

    /**
     * Get cache key for user
     */
    private function getCacheKey(int $userId): string
    {
        return $this->cachePrefix . $userId;
    }

    /**
     * Get user statistics
     */
    public function getUserStats(int $userId): array
    {
        $state = $this->getUserState($userId);
        
        return [
            'user_id' => $userId,
            'current_menu' => $state['current_menu'] ?? null,
            'context' => $state['context'] ?? null,
            'has_menu_data' => !empty($state['menu_data']),
            'has_context_data' => !empty($state['context_data']),
            'state_size' => count($state),
            'last_activity' => $state['last_activity'] ?? null
        ];
    }

    /**
     * Update last activity
     */
    public function updateLastActivity(int $userId): void
    {
        $this->updateUserState($userId, 'last_activity', now()->timestamp);
    }
}

<code php>
<?php

namespace App\Services\Telegram\Commands\Advanced;

use App\Contracts\LoggingServiceInterface;
use App\Services\Channels\TelegramChannel;
use App\Services\Telegram\Commands\Advanced\UserStateManager;
use Illuminate\Support\Facades\Http;

class AICommandService
{
    public function __construct(
        private LoggingServiceInterface $loggingService,
        private TelegramChannel $telegramChannel,
        private UserStateManager $userStateManager
    ) {}

    /**
     * Process AI chat message
     */
    public function processAIMessage(string $message, int $chatId, int $userId): array
    {
        try {
            $this->loggingService->logTelegramEvent('ai_message_processing_started', [
                'chat_id' => $chatId,
                'user_id' => $userId,
                'message_length' => strlen($message)
            ], 'info');

            // Get conversation context
            $context = $this->getConversationContext($userId);
            
            // Send typing indicator
            $this->telegramChannel->sendTypingIndicator((string) $chatId);

            // Process with AI
            $aiResponse = $this->getAIResponse($message, $context, $userId);

            if (!$aiResponse) {
                return [
                    'success' => false,
                    'message' => 'Desculpe, não consegui processar sua mensagem no momento. Tente novamente.'
                ];
            }

            // Save conversation to context
            $this->saveConversationContext($userId, $message, $aiResponse);

            // Send response
            $result = $this->telegramChannel->sendTextMessage($aiResponse, (string) $chatId);

            $this->loggingService->logTelegramEvent('ai_message_processed_successfully', [
                'chat_id' => $chatId,
                'user_id' => $userId,
                'response_length' => strlen($aiResponse)
            ], 'info');

            return $result;

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'process_ai_message',
                'message' => $message,
                'chat_id' => $chatId,
                'user_id' => $userId
            ]);

            return [
                'success' => false,
                'message' => 'Ocorreu um erro ao processar sua mensagem. Tente novamente.'
            ];
        }
    }

    /**
     * Get AI response from OpenAI
     */
    private function getAIResponse(string $message, array $context, int $userId): ?string
    {
        try {
            $apiKey = config('openai.api_key');
            
            if (!$apiKey) {
                $this->loggingService->logTelegramEvent('openai_api_key_missing', [], 'error');
                return null;
            }

            // Build conversation history
            $messages = [
                [
                    'role' => 'system',
                    'content' => $this->getSystemPrompt()
                ]
            ];

            // Add context if available
            if (!empty($context)) {
                foreach ($context as $conversation) {
                    $messages[] = [
                        'role' => 'user',
                        'content' => $conversation['user_message']
                    ];
                    $messages[] = [
                        'role' => 'assistant',
                        'content' => $conversation['ai_response']
                    ];
                }
            }

            // Add current message
            $messages[] = [
                'role' => 'user',
                'content' => $message
            ];

            $response = Http::withHeaders([
                'Authorization' => 'Bearer ' . $apiKey,
                'Content-Type' => 'application/json'
            ])->post('https://api.openai.com/v1/chat/completions', [
                'model' => config('openai.model', 'gpt-3.5-turbo'),
                'messages' => $messages,
                'max_tokens' => 500,
                'temperature' => 0.7
            ]);

            if ($response->successful()) {
                $data = $response->json();
                
                if (isset($data['choices'][0]['message']['content'])) {
                    return trim($data['choices'][0]['message']['content']);
                }
            }

            $this->loggingService->logTelegramEvent('openai_api_error', [
                'status' => $response->status(),
                'error' => $response->body()
            ], 'error');

            return null;

        } catch (\Exception $e) {
            $this->loggingService->logException($e, [
                'operation' => 'get_ai_response'
            ]);
            return null;
        }
    }

    /**
     * Get system prompt for AI
     */
    private function getSystemPrompt(): string
    {
        return "Você é um assistente virtual do sistema 'Rei do Óleo', especializado em ajudar com questões relacionadas a óleos automotivos e industriais. 

Você deve:
- Ser prestativo e educado
- Responder em português brasileiro
- Fornecer informações precisas sobre óleos
- Sugerir produtos quando apropriado
- Ser conciso mas informativo
- Manter um tom profissional mas amigável

Se não souber algo específico, admita e sugira onde o usuário pode encontrar mais informações.";
    }

    /**
     * Get conversation context
     */
    private function getConversationContext(int $userId): array
    {
        return $this->userStateManager->getUserStateValue($userId, 'ai_conversation_context', []);
    }

    /**
     * Save conversation context
     */
    private function saveConversationContext(int $userId, string $userMessage, string $aiResponse): void
    {
        $context = $this->getConversationContext($userId);
        
        // Add new conversation
        $context[] = [
            'user_message' => $userMessage,
            'ai_response' => $aiResponse,
            'timestamp' => now()->timestamp
        ];

        // Keep only last 10 conversations to avoid context overflow
        if (count($context) > 10) {
            $context = array_slice($context, -10);
        }

        $this->userStateManager->updateUserState($userId, 'ai_conversation_context', $context);
    }

    /**
     * Clear AI conversation context
     */
    public function clearAIContext(int $userId): void
    {
        $this->userStateManager->updateUserState($userId, 'ai_conversation_context', []);
        
        $this->loggingService->logTelegramEvent('ai_context_cleared', [
            'user_id' => $userId
        ], 'info');
    }

    /**
     * Get AI usage statistics
     */
    public function getAIStats(int $userId): array
    {
        $context = $this->getConversationContext($userId);
        
        return [
            'user_id' => $userId,
            'total_conversations' => count($context),
            'last_conversation' => !empty($context) ? end($context)['timestamp'] : null,
            'context_size' => count($context)
        ];
    }
}
</code>
